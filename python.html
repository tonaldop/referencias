🧩 Funções

def soma(a, b=0, *args, **kwargs):
    """Soma valores."""
    return a + b + sum(args)

# Chamadas
soma(1, 2)
soma(1, 2, 3, 4)
soma(1, b=5)

# Lambda

# Múltiplos retornos (tuple)
def divmod_(a, b):
    return a // b, a % b

💡 *args = lista de argumentos posicionais. **kwargs = dicionário de nomeados.

🏗️ Programação Orientada a Objetos

class Pessoa:
    # Atributo de classe
    especie = "humano"

    def __init__(self, nome, idade):
        self.nome = nome      # atributo de instância
        self.idade = idade

    def saudacao(self):
        return f"Olá, {self.nome}"

    def __str__(self):
        return f"Pessoa({self.nome}, {self.idade})"


# Herança
class Aluno(Pessoa):
    def __init__(self, nome, idade, matricula):
        super().__init__(nome, idade)
        self.matricula = matricula

    def saudacao(self):
        return f"Aluno {self.nome}"

⚠️ Tratamento de Exceções

try:
    x = int(input("Número: "))
    y = 10 / x
except ValueError:
    print("Não é número")
except ZeroDivisionError as e:
    print("Divisão por zero:", e)
except Exception as e:
    print("Erro:", e)
else:
    print("Tudo ok:", y)
finally:
    print("Sempre executa")

# Levantar exceção
raise ValueError("Valor inválido")

# Exceção customizada
class MinhaExcecao(Exception):
    pass
⚠️ Evite except: sozinho (captura tudo, inclusive Ctrl+C). Prefira capturar exceções específicas.

🎯 Recursos Avançados

Decoradores

def meu_decorador(func):
    def wrapper(*args, **kwargs):
        print("antes")
        resultado = func(*args, **kwargs)
        print("depois")
        return resultado
    return wrapper

@meu_decorador
def saudacao():
    print("Olá")

Geradores

def contador(n):
    for i in range(n):
        yield i

for x in contador(5):
    print(x)

# Generator expression (economiza memória)
quadrados = (x*x for x in range(1000000))

Context managers customizados

from contextlib import contextmanager

@contextmanager
def meu_contexto():
    print("entrando")
    yield
    print("saindo")

with meu_contexto():
    print("dentro")

Dataclasses (Python 3.7+)

from dataclasses import dataclass

@dataclass
class Pessoa:
    nome: str
    idade: int = 0

p = Pessoa("Ana", 30)
print(p)  # Pessoa(nome='Ana', idade=30)

🧪 Testes com pytest

# arquivo: test_soma.py
from minha_lib import soma

def test_soma_positivos():
    assert soma(2, 3) == 5

def test_soma_negativos():
    assert soma(-1, -1) == -2

# Rodar:
# pytest                    → todos os testes
# pytest -v                 → verboso
# pytest test_soma.py       → arquivo específico
# pytest -k "positivos"     → testes que casam com nome
# pytest --cov=minha_lib    → cobertura
Comando O que faz
pytest Roda todos os testes
pytest -v Modo verboso
pytest -x Para no primeiro erro
pytest -s Mostra prints
pytest --lf Roda só os que falharam antes
