🔀 Controle de Fluxo

if / else

if x > 10 {
    fmt.Println("maior")
} else if x == 10 {
    fmt.Println("igual")
} else {
    fmt.Println("menor")
}

for (único loop em Go)

for i := 0; i < 10; i++ { fmt.Println(i) }   // clássico
for i < 10 { ... }                          // tipo while
for { ... }                                 // infinito
for i, v := range lista { ... }             // range

switch

switch dia {
case 1, 7:
    fmt.Println("Fim de semana")
case 2, 3, 4, 5, 6:
    fmt.Println("Dia útil")
default:
    fmt.Println("Inválido")
}
💡 Dica: em Go não existe while — use for com condição única.

🧩 Funções

func soma(a, b int) int {
    return a + b
}

// Múltiplos retornos
func divmod(a, b int) (int, int) {
    return a / b, a % b
}

// Retornos nomeados
func divmod(a, b int) (q, r int) {
    q = a / b
    r = a % b
    return
}

// Variádica
func soma(nums ...int) int {
    total := 0
    for _, n := range nums { total += n }
    return total
}

🏗️ Structs e Métodos

type Pessoa struct {
    Nome string
    Idade int
}

// Método com receptor de valor
func (p Pessoa) Saudacao() string {
    return "Olá, " + p.Nome
}

// Método com receptor ponteiro (modifica)
func (p *Pessoa) Aniversario() {
    p.Idade++
}

p := Pessoa{Nome: "Ana", Idade: 30}
p.Aniversario()
fmt.Println(p.Saudacao(), p.Idade)

🔌 Interfaces

type Animal interface {
    Som() string
}

type Cachorro struct{}
func (c Cachorro) Som() string { return "Au au" }

type Gato struct{}
func (g Gato) Som() string { return "Miau" }

func fazerSom(a Animal) {
    fmt.Println(a.Som())
}
💡 Implicit interface: em Go, um tipo implementa uma interface apenas por ter os métodos — sem implements.

⚠️ Tratamento de Erros

// Erros são valores comuns
f, err := os.Open("arquivo.txt")
if err != nil {
    log.Fatal(err)
}
defer f.Close()

// Criando erros
return fmt.Errorf("falha ao processar: %w", err)

// Erros customizados
var ErrNaoEncontrado = errors.New("não encontrado")

// Verificando
if errors.Is(err, ErrNaoEncontrado) { ... }
⚠️ Go não tem try/catch. Use if err != nil sempre. panic/recover só em casos extremos.

🧵 Concorrência — Goroutines

// Goroutine: roda em paralelo
go minhaFuncao()

// Exemplo com WaitGroup
var wg sync.WaitGroup
for i := 0; i < 5; i++ {
    wg.Add(1)
    go func(n int) {
        defer wg.Done()
        fmt.Println(n)
    }(i)
}
wg.Wait()

Canais (channels)

ch := make(chan int)

go func() { ch <- 42 }()   // envia
valor := <-ch              // recebe

// Canal com buffer
ch := make(chan int, 3)

// Select (multiplexação)
select {
case v := <-ch1:
    fmt.Println("ch1:", v)
case v := <-ch2:
    fmt.Println("ch2:", v)
case <-time.After(time.Second):
    fmt.Println("timeout")
}
💡 Regra de ouro: nunca compartilhe memória — comunique-se por canais.

📂 Pacotes e Imports

import (

🧪 Testes

// arquivo: soma_test.go
package main

import "testing"

func TestSoma(t *testing.T) {
    got := soma(2, 3)
    want := 5
    if got != want {
        t.Errorf("soma(2,3) = %d; want %d", got, want)
    }
}

// Table-driven test (idiomático)
func TestSomaTabela(t *testing.T) {
    casos := []struct{ a, b, want int }{
        {1, 2, 3},
        {0, 0, 0},
        {-1, 1, 0},
    }
    for _, c := range casos {
        if got := soma(c.a, c.b); got != c.want {
            t.Errorf("soma(%d,%d) = %d; want %d", c.a, c.b, got, c.want)
        }
    }
}
Comando O que faz
go test Roda testes do pacote atual
go test -v Modo verboso
go test -cover Mostra cobertura
go test -run TestSoma Roda um teste específico
go test -bench=. Roda benchmarks

📁 Manipulação de Arquivos

// Ler arquivo inteiro
data, err := os.ReadFile("file.txt")
if err != nil { log.Fatal(err) }
fmt.Println(string(data))

// Escrever
err = os.WriteFile("out.txt", []byte("texto"), 0644)

// Ler linha por linha
f, _ := os.Open("file.txt")
defer f.Close()
sc := bufio.NewScanner(f)
for sc.Scan() {
    fmt.Println(sc.Text())
}

🌐 HTTP Server básico

package main

import (
    "fmt"
    "net/http"
)

func hello(w http.ResponseWriter, r *http.Request) {
    fmt.Fprintf(w, "Olá, %s!", r.URL.Query().Get("nome"))
}

func main() {
    http.HandleFunc("/hello", hello)
    http.ListenAndServe(":8080", nil)
}
